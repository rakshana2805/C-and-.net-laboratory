using System; 
class Number 
{ 
public int value; 
public Number(int v) 
{ 
} 
value = v; 
public static Number operator +(Number n1, Number n2) 
{ 
} 
return new Number(n1.value + n2.value); 
public static Number operator -(Number n1, Number n2) 
{ 
} 
return new Number(n1.value - n2.value); 
public static Number operator *(Number n1, Number n2) 
{ 
} 
return new Number(n1.value * n2.value); 
public static Number operator /(Number n1, Number n2) 
{ 
} 
return new Number(n1.value / n2.value); 
public void Display(string operation) 
{ 
} 
} 
Console.WriteLine(operation + " = " + value); 
class Program 
{ 
static void Main(string[] args) 
{ 
Number num1 = new Number(20); 
Number num2 = new Number(10); 
Console.WriteLine("First Number  : " + num1.value); 
Console.WriteLine("Second Number : " + num2.value); 
Number add = num1 + num2; 
Number sub = num1 - num2; 
Number mul = num1 * num2; 
Number div = num1 / num2; 
Console.WriteLine("\nResults After Operator Overloading"); 
add.Display("Addition"); 
sub.Display("Subtraction"); 
mul.Display("Multiplication"); 
div.Display("Division"); 
Console.ReadLine(); 
} 
} 


OUTPUT : 
First Number  : 20 
Second Number : 10 
Results After Operator Overloading 
Addition = 30 
Subtraction = 10 
Multiplication = 200 
Division = 2
